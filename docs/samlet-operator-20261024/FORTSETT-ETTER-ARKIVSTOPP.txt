def load(p):
    return json.loads(Path(p).read_text(encoding='utf-8-sig'))


def save(p,value):
    with Path(p).open('x',encoding='utf-8') as f:
        json.dump(value,f,ensure_ascii=False,indent=2,allow_nan=False)
    Path(p).chmod(0o600)


def command(args,*,env=None,input=None):
    r=subprocess.run([str(a) for a in args],input=input,capture_output=True,env=env,timeout=1800)
    if r.returncode:
        raise RuntimeError('Operator dependency failed: '+Path(str(args[0])).name)
    return r.stdout

def quiescent(config):
    if os.getenv('ENABLE_WEB_BACKGROUND_SERVICES','false').lower() in {'true','yes','1','on'}:
        raise ValueError('Web background workers enabled; no quiescence claim is made')
    cs=connections(config)
    try:
        for c in cs.values():
            c.set_session(readonly=True)
            with c.cursor() as q:
                q.execute("SELECT count(*) FROM pg_stat_activity WHERE datname=current_database() AND pid<>pg_backend_pid() AND state IN ('active','idle in transaction','idle in transaction (aborted)')")
                if q.fetchone()[0]:raise ValueError('Another active/open transaction exists; archive stopped')
            c.rollback()
    finally:
        for c in cs.values():c.rollback();c.close()


def tools(work):

    found={}
    for name in ('pg_dump','pg_restore','initdb','pg_ctl','postgres'):
        candidate=next((p/name for p in paths if (p/name).is_file()),None)
        candidate=candidate or shutil.which(name)
        if not candidate:raise ValueError('Missing PostgreSQL 18 dependency: '+name)
        candidate=Path(candidate)
        version=command([candidate,'--version']).decode()
        if not __import__('re').search(r'\b18\.',version):raise ValueError('PostgreSQL 18 tools required')

    return found

def rollback_close(connection):
    """A broken connection during cleanup must not replace the original error."""
    failure_pending=sys.exc_info()[0] is not None
    cleanup_error=None
    try:
        if not connection.closed:connection.rollback()
    except Exception as error:cleanup_error=error
    try:connection.close()
    except Exception as error:
        if cleanup_error is None:cleanup_error=error
    if cleanup_error is not None and not failure_pending:raise cleanup_error


class FingerprintChildError(RuntimeError):
    def __init__(self,failure):
        super().__init__('Read-only fingerprint child failed; original cause preserved')
        self.fingerprint_failure=failure


class OneRowCursor:
    """No thousand-row libpq batches; preserve the core canonical JSON algorithm."""
    def __init__(self,cursor,progress):
        self.progress=progress
        self.progress.update(table_index=self.progress.get('table_index',0)+1,rows_fetched_this_table=0)
        self.cursor=cursor
        self.cursor.itersize=1
        self.cursor.arraysize=1
    @property
    def itersize(self):return 1
    @itersize.setter
    def itersize(self,value):self.cursor.itersize=1
    def execute(self,*args,**kwargs):
        self.progress['phase']='EXECUTE_NAMED_CURSOR'
        return self.cursor.execute(*args,**kwargs)
    def __enter__(self):
        self.cursor.__enter__()
        return self
    def __exit__(self,*args):
        try:return self.cursor.__exit__(*args)
        except Exception:
            if args[0] is None:raise
            return False
    def __iter__(self):
        while True:
            self.progress['phase']='FETCH_ONE_ROW'
            row=self.cursor.fetchone()
            if row is None:return
            self.progress['rows_fetched_this_table']+=1
            self.progress['rows_fetched_total']=self.progress.get('rows_fetched_total',0)+1
            self.progress['last_row_characters']=len(row[0])
            self.progress['max_row_characters']=max(self.progress.get('max_row_characters',0),len(row[0]))
            self.progress['phase']='CANONICALIZE_AND_HASH_ROW'
            yield row


class OneRowConnection:
    def __init__(self,connection,progress=None):
        self.connection=connection;self.progress=progress if progress is not None else {}
    def cursor(self,*args,**kwargs):
        result=self.connection.cursor(*args,**kwargs)
        return OneRowCursor(result,self.progress) if kwargs.get('name') or (args and args[0]) else result


def require_isolated_fingerprint(parameters,work):
    """Production row_to_json scans caused server SIGKILL; never retry them."""
    socket=Path(work).resolve()/'pgsocket'
    host=parameters.get('host','')
    if (parameters.get('hostaddr') or not host or not Path(host).is_absolute()

        or not re.fullmatch(r'nok_restore_[0-9]+',parameters.get('dbname',''))):
        raise ValueError('SOURCE FINGERPRINT SUSPENDED: PostgreSQL SIGKILL during archive FETCH; only the private isolated restore target may be scanned')


def fingerprint_child(work):
    """Lightweight read-only child, attached to the caller's exact snapshot."""
    if os.getenv('APP_ENVIRONMENT')!='production':raise ValueError('Actual production environment required')
    expected=load(work/'source-manifest.json')['core_sources']['services/simulation_archive.py']
    if hashlib.sha256((ROOT/'services/simulation_archive.py').read_bytes()).hexdigest()!=expected:
        raise ValueError('Core fingerprint implementation differs')
    request=json.load(sys.stdin)
    require_isolated_fingerprint(request['parameters'],work)
    snapshot=request['snapshot']
    if not re.fullmatch(r'[0-9A-Fa-f]+-[0-9A-Fa-f]+-[0-9]+',snapshot):raise ValueError('Invalid exported snapshot')
    import psycopg2
    from services.simulation_archive import database_fingerprint
    connection=psycopg2.connect(**request['parameters'])
    progress={'phase':'IMPORT_SNAPSHOT'}
    try:
        connection.set_session(readonly=True,isolation_level='REPEATABLE READ',autocommit=False)
        with connection.cursor() as cursor:
            cursor.execute('SET TRANSACTION SNAPSHOT %s',(snapshot,))
            cursor.execute('SHOW transaction_read_only')
            if cursor.fetchone()[0]!='on':raise ValueError('Fingerprint must remain read-only')
        value=database_fingerprint(OneRowConnection(connection,progress),ignore_period=request.get('ignore_period'))
    except Exception as error:
        error.operator_fingerprint_progress=dict(progress)
        raise
    finally:
        rollback_close(connection)
    print(json.dumps(dict(status='FINGERPRINT_VERIFIED',fingerprint=value,peak_rss_kib=resource.getrusage(resource.RUSAGE_SELF).ru_maxrss)))


def install_bounded_fingerprints(archive,work):
    def fingerprint(connection,*,ignore_period=None):
        require_isolated_fingerprint(connection.get_dsn_parameters(),work)
        # Export within the original open transaction. pg_dump and the child
        # see identical data; no snapshot is guessed or replaced with live data.
        with connection.cursor() as cursor:
            cursor.execute('SELECT pg_export_snapshot()')
            snapshot=cursor.fetchone()[0]
        parameters=dict(connection.get_dsn_parameters())
        parameters['connect_timeout']='15'
        if connection.info.password:parameters['password']=connection.info.password
        request=dict(snapshot=snapshot,parameters=parameters,ignore_period=ignore_period)
        # Credentials are passed only through the child's private stdin pipe.
        result=subprocess.run([sys.executable,str(Path(__file__).resolve()),'fingerprint-readonly','--work',str(work)],
            input=json.dumps(request).encode(),capture_output=True,timeout=1800)
        try:reply=json.loads(result.stdout)
        except (ValueError,UnicodeDecodeError):raise RuntimeError('Fingerprint child failed without a valid result; no archive approval') from None
        if result.returncode or reply.get('status')!='FINGERPRINT_VERIFIED':
            raise FingerprintChildError(reply)
        print('Read-only fingerprint verified: tables='+str(len(reply['fingerprint']))+'; child peak RSS KiB='+str(reply['peak_rss_kib']),flush=True)
        return reply['fingerprint']
    archive.database_fingerprint=fingerprint

def sanitized_failure(error):
    """Useful SQL diagnostics without DSNs, passwords, connection identities or rows."""
    text=str(error);redactions=[]
    for variable,dsn in os.environ.items():
        if variable not in {'DATABASE_URL','AUTH_DATABASE_URL','POSTGRES_URL','RENDER_DATABASE_URL'} and not dsn.startswith(('postgres://','postgresql://')):continue
        if not dsn:continue
        redactions.append(dsn)
        if dsn.startswith(('postgres://','postgresql://')):
            parsed=urlsplit(dsn)
            redactions.extend(unquote(value) for value in (parsed.username,parsed.password,parsed.hostname,parsed.path.lstrip('/')) if value)
        try:
            from psycopg2.extensions import parse_dsn
            redactions.extend(str(value) for key,value in parse_dsn(dsn).items() if key not in {'port','connect_timeout'} and value)
        except Exception:pass
    for value in sorted(set(redactions),key=len,reverse=True):text=text.replace(value,'[redacted]')
    text=re.sub(r'(?i)postgres(?:ql)?://[^\s]+','[redacted DSN]',text)
    result=dict(error_type=type(error).__name__,sqlstate=getattr(error,'pgcode',None),reason=text[:1800],
                frames=[dict(file=Path(frame.filename).name,function=frame.name,line=frame.lineno)
                        for frame in traceback.extract_tb(error.__traceback__)][-10:])
    if hasattr(error,'operator_fingerprint_progress'):result['progress']=error.operator_fingerprint_progress
    if isinstance(error,FingerprintChildError):result['original_failure']=error.fingerprint_failure
    return result

# No activation, deploy, paid resource creation or execution-policy change.
$script:AksjeLastTransportAt = [datetime]::MinValue
function Invoke-AksjeNative {
    param([string]$Program,[string[]]$NativeArguments,[string]$LogDirectory,
          [switch]$Transport,[switch]$CaptureOutput,[int[]]$RetryDelays=@(8,16,32),[int]$MinimumGapMs=2000)
    $null=Get-Command -Name $Program -CommandType Application -ErrorAction Stop
    $Attempts = 1
    if ($Transport) { $Attempts += $RetryDelays.Count }
    for ($Attempt=0; $Attempt -lt $Attempts; $Attempt++) {
        if ($Transport) {
            $Gap = $MinimumGapMs - ([datetime]::UtcNow-$script:AksjeLastTransportAt).TotalMilliseconds
            if ($Gap -gt 0) { Start-Sleep -Milliseconds ([int]$Gap) }
        }
        $Log = Join-Path $LogDirectory ('native-'+[guid]::NewGuid().ToString('N')+'.log')
        $SavedPreference=$ErrorActionPreference; $ExitCode=255; $Output=@()
        try {
            $ErrorActionPreference='Continue'
            if ($CaptureOutput) { $Output=@(& $Program @NativeArguments 2> $Log) }
            else { & $Program @NativeArguments 2> $Log | Out-Host }
            $ExitCode=$LASTEXITCODE
        } finally {
            $ErrorActionPreference=$SavedPreference
            if ($Transport) { $script:AksjeLastTransportAt=[datetime]::UtcNow }
        }
        $ErrorText=''
        if (Test-Path -LiteralPath $Log) { $ErrorText=[System.IO.File]::ReadAllText($Log) }
        $Denied = $ErrorText -match '(?i)Permission denied|Host key verification failed|REMOTE HOST IDENTIFICATION HAS CHANGED'
        # Retry only failures explicitly reported BEFORE SSH key exchange.
        # Never repeat a remote operation after an ambiguous mid-session disconnect.
        $BeforeLogin = $Transport -and -not $Denied -and ($ExitCode -in @(1,255)) -and
            ($ErrorText -match '(?i)kex_exchange_identification:.*(?:Connection reset|Connection closed)' -or
             $ErrorText -match '(?i)ssh: connect to host .* port [0-9]+: (?:Connection timed out|Connection refused|No route to host)')
        if ($ExitCode -ne 0 -and $BeforeLogin -and $Attempt -lt ($Attempts-1)) {
            $Delay=$RetryDelays[$Attempt]
            Write-Host "SSH-forbindelsen ble brutt for innlogging. Venter $Delay sekunder og bruker samme nokkel og jobb."
            if ($Delay -gt 0) { Start-Sleep -Seconds $Delay }
            continue
        }
        return [pscustomobject]@{ExitCode=$ExitCode;Output=$Output;ErrorText=$ErrorText;Log=$Log;BeforeLogin=$BeforeLogin}
    }
}

function Get-AksjeResumeContext([string]$OperatorRoot,[string]$JobId,[string]$ExpectedService,[switch]$InstallerCheckpoint) {
    if ($JobId -cnotmatch '^nok-job-[0-9]{8}T[0-9]{6}Z$') { throw 'Ugyldig jobb-ID; ingen handling startet' }
    $Folder=Join-Path $OperatorRoot $JobId
    $Location=Get-Content -LiteralPath (Join-Path $Folder 'job-location.json') -Raw | ConvertFrom-Json
    $Plan=Get-Content -LiteralPath (Join-Path $Folder 'plan.json') -Raw | ConvertFrom-Json
    if ($Location.remote -cne ('/tmp/'+$JobId) -or $Location.service -cne $ExpectedService -or
        [IO.Path]::GetFullPath($Location.local) -ne [IO.Path]::GetFullPath($Folder) -or
        $Plan.job_id -cne $JobId -or $Plan.activation_authorized -isnot [bool] -or $Plan.activation_authorized -ne $false -or
        $Plan.approval_A_already_granted -isnot [bool] -or $Plan.approval_A_already_granted -ne $true) {
        throw 'Lagret jobbidentitet avviker; ingen automatisk gjenopptakelse'
    }
    $Forbidden=@('archive-receipt.json','preparation-attempt.json','prepared.json','final-result.json','config.json','archive-key.pem','independent-archives','isolated-pg','archives')
    if (-not $InstallerCheckpoint) { $Forbidden+=@('source-attestation.json') }
    foreach ($Name in $Forbidden) {
        if (Test-Path -LiteralPath (Join-Path $Folder $Name)) { throw "Jobben er kommet lenger enn forhandskontroll ($Name). Ingen fase gjentas automatisk." }
    }
    if ($InstallerCheckpoint) { Test-AksjeSourceAttestation $Folder }
    return [pscustomobject]@{Folder=$Folder;Location=$Location;Plan=$Plan}
}
function Test-AksjeSourceAttestation([string]$Folder) {
    $Attestation=Get-Content -LiteralPath (Join-Path $Folder 'source-attestation.json') -Raw | ConvertFrom-Json

    if ($Attestation.all_sources_reviewed_and_writers_stopped -isnot [bool] -or
        $Attestation.all_sources_reviewed_and_writers_stopped -ne $true -or
        [string]$Attestation.inventory_sha256 -cnotmatch '^[a-f0-9]{64}$' -or
        $Attestation.inventory_sha256 -cne $Preflight.inventory_sha256) {
        throw 'Tidligere kildeattestasjon er ugyldig eller tilhorer et annet inventar; ingen fortsettelse'
    }
}
function Get-AksjeExistingSshKey([string]$OperatorRoot) {
    if (-not (Test-Path -LiteralPath $OperatorRoot)) { return $null }
    $Candidate = Get-ChildItem -LiteralPath $OperatorRoot -Filter 'render-operator-ssh' -File -Recurse -ErrorAction Stop |
        Sort-Object LastWriteTime -Descending | Select-Object -First 1

function Get-AksjePublicSshKey([string]$SshKey) {

        $Preflight | Format-List
        $Drive = Get-PSDrive -Name ([System.IO.Path]::GetPathRoot($LocalJob).Substring(0,1))
        if ($Drive.Free -lt (2 * [double]$Preflight.source_bytes)) { throw 'For lite ledig plass til uavhengig arkivkopi paa PC-en' }
        if ($ResumeInstaller -or $ResumeArchive) {
            Download 'source-attestation.json'
            Test-AksjeSourceAttestation $LocalJob
            Write-Host 'Tidligere kildeattestasjon er kontrollert og bevart. Hold fortsatt scheduler og andre skrivere stanset.'
        } else {
            Write-Host 'Kontroller at alle relevante tjenester, gamle/eksterne lagre er inkludert, cron er suspendert og alle skrivere er stanset. Hold app-faner lukket. Ikke bekreft hvis dette er ukjent.'
            $Ack = Read-Host 'Skriv KILDENE ER KOMPLETTE OG SKRIVERNE STANSET'
            if ($Ack -ne 'KILDENE ER KOMPLETTE OG SKRIVERNE STANSET') { throw 'Ingen arkivering uten bekreftet kildedekning og skriverstans' }
            Write-NewUtf8 "$LocalJob\source-attestation.json" (@{inventory_sha256=$Preflight.inventory_sha256;all_sources_reviewed_and_writers_stopped=$true;actor=$Identity;at=[DateTime]::UtcNow.ToString('o')} | ConvertTo-Json)
            Upload "$LocalJob\source-attestation.json" 'source-attestation.json'
        }
        if (-not [string]::IsNullOrWhiteSpace($InstallerText)) {
            $FixedInstaller=Join-Path $LocalJob ('install-pg18-corrected-'+[guid]::NewGuid().ToString('N')+'.sh')
            Write-NewUtf8 $FixedInstaller ($InstallerText.Replace("`r`n","`n")+"`n")
            Upload $FixedInstaller 'install_pg18.sh'
        }
        if ($ResumeArchive) {
            Write-Host '[2/5] Kontrollerer mislykket, uforseglet arkivsteg. Gjenbruker installerte PG18-verktoy.'
            $ActiveStage='archive-checkpoint'
            Remote "python $WorkerScript archive-checkpoint --work $RemoteJob"
        } else {
            Write-Host '[2/5] Installerer isolerte PG18-verktoy i arbeidsmappen, ingen ny betalt ressurs'
            $ActiveStage='install'
            Remote "NOK_WORK=$RemoteJob bash $RemoteJob/install_pg18.sh"
        }
        $SensitivePhaseEntered=$true
        Upload "$KeysFolder\archive-key.pem" 'archive-key.pem'
        Remote "chmod 600 $RemoteJob/archive-key.pem"
        Write-Host '[3/5] Komplett kryptert sluttarkiv'
        $ActiveStage='archive'
        Remote "python $WorkerScript archive --work $RemoteJob"
        Download 'archive-receipt.json'
        $Receipt = Get-Content "$LocalJob\archive-receipt.json" -Raw | ConvertFrom-Json
        $ArchiveId = [string]$Receipt.archive_id
        if ($ArchiveId -notmatch '^[A-Za-z0-9-]+$') { throw 'Ugyldig arkiv-ID' }
        $Archives = Join-Path $LocalJob 'independent-archives'
        New-Item -ItemType Directory $Archives | Out-Null
        Native 'scp.exe' ($ScpOptions + @('-r',"${RenderSsh}:$RemoteJob/archives/$ArchiveId",$Archives))
        $Cipher = Join-Path $Archives "$ArchiveId\archive.cms"
        $CipherHash = (Get-FileHash $Cipher -Algorithm SHA256).Hash.ToLowerInvariant()
        if ($CipherHash -ne $Receipt.cipher_sha256) { throw 'Uavhengig Windows-kopi har feil kontrollsum' }
        Native $OpenSSL @('cms','-decrypt','-binary','-inform','DER','-in',$Cipher,'-recip',"$KeysFolder\archive-certificate.pem",'-inkey',"$KeysFolder\archive-key.pem",'-out','NUL')
        $CopyProof = @{archive_id=$ArchiveId;cipher_sha256=$CipherHash;verified_windows_copy=$true;local_decryption_verified=$true;copy_kind='WINDOWS_OFFSITE_VERIFIED_SCP_COPY';verified_at=[DateTime]::UtcNow.ToString('o')}
        Write-NewUtf8 "$LocalJob\windows-copy-proof.json" ($CopyProof | ConvertTo-Json)
        Upload "$LocalJob\windows-copy-proof.json" 'windows-copy-proof.json'
        Write-Host '[4/5] Fysisk restore paa eksisterende arbeidsplass og opprettelse av INAKTIVE DRAFT-kontoer'
        $ActiveStage='restore-drafts'
        Remote "python $WorkerScript restore-drafts --work $RemoteJob"
        Write-Host '[5/5] Henter faktiske kvitteringer og samlet status'
        foreach ($Name in @('final-result.json','restore-result.json','prepared.json','account-settings-for-review.json','inventory.json','source-databases.json','config.json')) { Download $Name }
        $Result = Get-Content "$LocalJob\final-result.json" -Raw | ConvertFrom-Json
        $Result | Format-List
        Write-Host "Ferdig med denne avgrensede kjoringen. Privat arkiv og kvitteringer: $LocalJob"
        Write-Host 'Ingen konto er aktivert. B er ikke klar: faktiske kostnader, livebevis og signerte grants maa dokumenteres.'
        Write-Host 'Del bare final-result.json som status, ikke arkivet, private nokler eller hele produksjonskonfigurasjonen.'
    }
    catch {
        $Failure = $_
        $RemoteStateUncertain=$Failure.Exception.Data.Contains('AksjeConnectionUncertain') -and [bool]$Failure.Exception.Data['AksjeConnectionUncertain']
        $BeforeLoginFailure=$Failure.Exception.Data.Contains('AksjeBeforeLogin') -and [bool]$Failure.Exception.Data['AksjeBeforeLogin']
        if (-not $BeforeLoginFailure -and -not $RemoteStateUncertain) {
            if ($ActiveStage -eq 'install') {
                $InstallLogName='pg18-install-'+[guid]::NewGuid().ToString('N')+'.log'
                try { Download 'pg18-install.log' $InstallLogName } catch { }
                if (Test-Path -LiteralPath (Join-Path $LocalJob $InstallLogName)) {
                    Write-Host 'Siste linjer fra den faktiske installasjonsloggen:'
                    Get-Content -LiteralPath (Join-Path $LocalJob $InstallLogName) -Tail 35 | Out-Host
                }
            } else { try { Download "STOPPED-$ActiveStage.json" } catch { } }
        }
        if ($RemoteStateUncertain) { Write-Warning 'Tilkoblingen ble brutt etter at fjernoperasjonen kan ha startet. Fjernstatus er ukjent; ingen automatisk gjentakelse eller opprydding.' }
        Write-Host "Jobben er stoppet uten aktivering. Eventuelle private STOPPED-kvitteringer ligger i $LocalJob"
        throw $Failure
    }
    finally {
        if ('AksjeNok.PowerState' -as [type]) { [void][AksjeNok.PowerState]::SetThreadExecutionState([uint32]2147483648) }
        if ($RemoteCreated -and $SensitivePhaseEntered -and -not $RemoteStateUncertain) {
            try {
                Native 'ssh.exe' ($SshOptions+@($RenderSsh,"python $WorkerScript cleanup --work $RemoteJob"))
            } catch { Write-Warning 'Midlertidig arkivnokkel/restore-prosess maa kontrolleres i den private jobbmappen. Originalnokkelen paa PC-en bevares.' }
        }
        Write-Host "Jobbmappe: $LocalJob"
        Write-Host 'Ingen automatisk ny start, sletting av historikk eller overgang til annen periode.'
    }
}
function Resume-AksjeOperator {
    param([string]$JobId='', [switch]$ResumeInstaller, [string]$InstallerText='', [switch]$ResumeArchive, [string]$WorkerText='')
    if ([string]::IsNullOrWhiteSpace($JobId)) {
        $OperatorRoot=Join-Path $env:LOCALAPPDATA 'AksjeNokOperator'
        $Candidates=@(Get-ChildItem -LiteralPath $OperatorRoot -Directory -Filter 'nok-job-*' -ErrorAction Stop | Where-Object {
            $Folder=$_.FullName
            (Test-Path -LiteralPath (Join-Path $Folder 'job-location.json')) -and
            (Test-Path -LiteralPath (Join-Path $Folder 'plan.json')) -and
            ((($ResumeInstaller -or $ResumeArchive) -and (Test-Path -LiteralPath (Join-Path $Folder 'source-attestation.json'))) -or
             (-not ($ResumeInstaller -or $ResumeArchive) -and -not (Test-Path -LiteralPath (Join-Path $Folder 'source-attestation.json')))) -and
            ((-not $ResumeArchive) -or (Test-Path -LiteralPath (Join-Path $Folder 'STOPPED-archive.json'))) -and
            -not (Test-Path -LiteralPath (Join-Path $Folder 'archive-receipt.json')) -and
            -not (Test-Path -LiteralPath (Join-Path $Folder 'preparation-attempt.json')) -and
            -not (Test-Path -LiteralPath (Join-Path $Folder 'prepared.json')) -and
            -not (Test-Path -LiteralPath (Join-Path $Folder 'final-result.json'))
        })
        if ($Candidates.Count -eq 1) { $JobId=$Candidates[0].Name }
        else { $JobId=Read-Host 'Jobb-ID fra feilmeldingen som skal fortsettes (nok-job-...)' }
    }
    Start-AksjeOperator -ResumeJob $JobId.Trim() -ResumeInstaller:$ResumeInstaller -InstallerText $InstallerText -ResumeArchive:$ResumeArchive -WorkerText $WorkerText
}
Resume-AksjeOperator -ResumeArchive -WorkerText $script:AksjeCorrectedArchiveWorker
