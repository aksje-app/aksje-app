# No activation, deploy, paid resource creation or execution-policy change.
function Get-AksjeExistingSshKey([string]$OperatorRoot) {
    if (-not (Test-Path -LiteralPath $OperatorRoot)) { return $null }
    $Candidate = Get-ChildItem -LiteralPath $OperatorRoot -Filter 'render-operator-ssh' -File -Recurse -ErrorAction Stop |
        Sort-Object LastWriteTime -Descending | Select-Object -First 1
    if ($null -eq $Candidate) { return $null }
    return $Candidate.FullName
}

function Get-AksjePublicSshKey([string]$SshKey) {
    $DerivedPublic = & ssh-keygen.exe -y -f $SshKey
    if ($LASTEXITCODE -ne 0) { throw 'Kunne ikke lese den offentlige SSH-identiteten' }
    $PublicFields = ([string]($DerivedPublic | Select-Object -First 1)).Trim() -split '\s+'
    if ($PublicFields.Count -lt 2) { throw 'Ufullstendig offentlig SSH-nokkel' }
    $PublicSsh = $PublicFields[0] + ' ' + $PublicFields[1] + ' aksje-nok-operator'
    if ($PublicSsh -notmatch '^ssh-ed25519\s+[A-Za-z0-9+/=]+\s+aksje-nok-operator$') { throw 'Ugyldig offentlig SSH-nokkel; ingen registrering eller produksjonsjobb startet' }
    return $PublicSsh
}

function Start-AksjeOperator {
    $ErrorActionPreference = 'Stop'
    Set-StrictMode -Version Latest
    $Utf8 = New-Object System.Text.UTF8Encoding($false)
    $JobId = 'nok-job-' + [DateTime]::UtcNow.ToString('yyyyMMddTHHmmssZ')
    $LocalJob = Join-Path $env:LOCALAPPDATA ('AksjeNokOperator\' + $JobId)
    $RemoteJob = '/tmp/' + $JobId
    $RenderSsh = 'srv-d7obt5e47okc739susgg@ssh.ohio.render.com'
    $Package = Read-Host 'Mappe der den samlede operatorpakken er pakket ut'
    $KeysFolder = Read-Host 'Lokal mappe med archive-certificate.pem, archive-key.pem og operator-certificate.pem (ikke lim inn filinnhold)'
    $Required = @('render_batch.py','install_pg18.sh','source-manifest.json')
    $Missing = New-Object System.Collections.Generic.List[string]
    foreach ($Name in $Required) { if (-not (Test-Path -LiteralPath (Join-Path $Package $Name))) { $Missing.Add("Pakkefil: $Name") } }
    foreach ($Name in @('archive-certificate.pem','archive-key.pem','operator-certificate.pem')) {
        if (-not (Test-Path -LiteralPath (Join-Path $KeysFolder $Name))) { $Missing.Add("Lokal fil: $Name") }
    }
    foreach ($Tool in @('ssh.exe','scp.exe','ssh-keygen.exe')) { if (-not (Get-Command $Tool -ErrorAction SilentlyContinue)) { $Missing.Add("Windows OpenSSH: $Tool") } }
    $OpenSSL = $null
    $OpenSSLCommand = Get-Command openssl.exe -ErrorAction SilentlyContinue
    if ($OpenSSLCommand) { $OpenSSL = $OpenSSLCommand.Source }
    if (-not $OpenSSL) {
        foreach ($Candidate in @('C:\Program Files\Git\usr\bin\openssl.exe','C:\Program Files\PostgreSQL\18\bin\openssl.exe')) {
            if (Test-Path -LiteralPath $Candidate) { $OpenSSL = $Candidate; break }
        }
    }
    if (-not $OpenSSL) { $Missing.Add('Lokal OpenSSL') }
    if ($Missing.Count) { throw ('Samlet forhandskontroll stoppet. Mangler: '+($Missing -join '; ')+'. Ingen produksjonsoperasjon startet. Ikke send private filer i chatten.') }
    if (Test-Path -LiteralPath $LocalJob) { throw 'Jobbmappen finnes allerede; ingen automatisk overskriving eller ny start.' }
    New-Item -ItemType Directory -Path $LocalJob | Out-Null
    $Identity = [System.Security.Principal.WindowsIdentity]::GetCurrent().Name
    & icacls.exe $LocalJob /inheritance:r /grant:r "${Identity}:(OI)(CI)F" | Out-Null
    if ($LASTEXITCODE -ne 0) { throw 'Kunne ikke opprette privat lokal jobbmappe' }
    function Write-NewUtf8([string]$Path, [string]$Text) {
        $Stream = [System.IO.File]::Open($Path,[System.IO.FileMode]::CreateNew,[System.IO.FileAccess]::Write,[System.IO.FileShare]::None)
        try { $Bytes=$Utf8.GetBytes($Text); $Stream.Write($Bytes,0,$Bytes.Length) } finally { $Stream.Dispose() }
    }
    function Native([string]$Program, [string[]]$NativeArguments) {
        & $Program @NativeArguments
        if ($LASTEXITCODE -ne 0) { throw "Operasjonen stoppet: $Program (exit $LASTEXITCODE). Ingen automatisk omgaaelse." }
    }
    # Validate the existing archive key locally. Never read operator-key.pem.
    Native $OpenSSL @('x509','-in',"$KeysFolder\archive-certificate.pem",'-pubkey','-noout','-out',"$LocalJob\archive-public-from-cert.pem")
    Native $OpenSSL @('pkey','-in',"$KeysFolder\archive-key.pem",'-passin','pass:','-pubout','-out',"$LocalJob\archive-public-from-key.pem")
    if ((Get-FileHash "$LocalJob\archive-public-from-cert.pem").Hash -ne (Get-FileHash "$LocalJob\archive-public-from-key.pem").Hash) { throw 'Arkivnokkel og sertifikat tilhorer ikke samme par' }
    Native $OpenSSL @('x509','-in',"$KeysFolder\operator-certificate.pem",'-noout')
    # Reuse the latest dedicated job identity instead of changing keys on retry.
    $SshKey = Get-AksjeExistingSshKey (Join-Path $env:LOCALAPPDATA 'AksjeNokOperator')
    if (-not $SshKey) {
        $SshKey = Join-Path $LocalJob 'render-operator-ssh'
        $Info = New-Object System.Diagnostics.ProcessStartInfo
        $Info.FileName = (Get-Command ssh-keygen.exe).Source
        $Info.UseShellExecute = $false
        $Info.Arguments = '-q -t ed25519 -N "" -f "' + $SshKey + '" -C "aksje-nok-operator"'
        $Process = [System.Diagnostics.Process]::Start($Info)
        $Process.WaitForExit()
        if ($Process.ExitCode -ne 0) { throw 'Kunne ikke opprette egen lokal SSH-identitet' }
    }
    Write-Host "Bruker samme eksisterende SSH-identitet: $SshKey"
    # Derive the public key from the actual SSH identity; never touch operator-key.pem.
    $PublicSsh = Get-AksjePublicSshKey $SshKey
    Write-NewUtf8 "$LocalJob\render-operator-used.pub" ($PublicSsh + "`n")
    $SshOptions = @('-i',$SshKey,'-o','IdentitiesOnly=yes','-o','StrictHostKeyChecking=ask','-o','PasswordAuthentication=no','-o','KbdInteractiveAuthentication=no')
    $ScpOptions = @('-O','-i',$SshKey,'-o','IdentitiesOnly=yes','-o','StrictHostKeyChecking=yes','-o','PasswordAuthentication=no','-o','KbdInteractiveAuthentication=no')
    $AuthSucceeded = $false
    $SavedErrorPreference = $ErrorActionPreference
    try {
        $ErrorActionPreference = 'Continue'
        $AuthOutput = & ssh.exe @SshOptions -o BatchMode=yes $RenderSsh true 2>&1
        $AuthSucceeded = $LASTEXITCODE -eq 0
    } catch { $AuthSucceeded = $false }
    finally { $ErrorActionPreference = $SavedErrorPreference }
    if (-not $AuthSucceeded) {
        Write-Host "`nLegg denne OFFENTLIGE SSH-nokkelen i Render Account Settings > SSH Public Keys:"
        Write-Host $PublicSsh
        Native 'ssh-keygen.exe' @('-lf',"$LocalJob\render-operator-used.pub")
        if (Get-Command Set-Clipboard -ErrorAction SilentlyContinue) { Set-Clipboard -Value $PublicSsh }
        Write-Host 'Hele den offentlige nokkelen er kopiert hvis utklippstavlen er tilgjengelig. Ikke del privat nokkel.'
        Start-Process 'https://dashboard.render.com/u/settings'
        [void](Read-Host 'Trykk Enter etter at akkurat denne offentlige nokkelen er registrert i Render')
    } else { Write-Host 'Eksisterende SSH-tilgang fungerer; ingen ny nokkelregistrering nodig.' }
    Write-Host 'Forste SSH-tilkobling krever normal kontroll av Render-serverens vertsnoekkel. Ikke godta ukjent/avvikende fingeravtrykk.'
    Native 'ssh.exe' ($SshOptions + @($RenderSsh,'true'))
    function Remote([string]$Text) { Native 'ssh.exe' ($SshOptions + @($RenderSsh,$Text)) }
    function Upload([string]$Source,[string]$Name) { Native 'scp.exe' ($ScpOptions + @($Source,"${RenderSsh}:$RemoteJob/$Name")) }
    function Download([string]$Name) {
        $Destination = Join-Path $LocalJob $Name
        if (Test-Path -LiteralPath $Destination) { throw "Lokal kvittering finnes allerede: $Name" }
        Native 'scp.exe' ($ScpOptions + @("${RenderSsh}:$RemoteJob/$Name",$Destination))
    }
    $Plan = @{actor=$Identity;approval_A_already_granted=$true;activation_authorized=$false;job_id=$JobId}
    Write-NewUtf8 "$LocalJob\plan.json" ($Plan | ConvertTo-Json -Depth 10)
    Write-NewUtf8 "$LocalJob\job-location.json" (@{remote=$RemoteJob;service=$RenderSsh;local=$LocalJob} | ConvertTo-Json)
    Remote "umask 077 && mkdir -m 700 $RemoteJob"
    $RemoteCreated = $true
    try {
        # Request wakefulness for this job only; do not change Windows power policies.
        if (-not ('AksjeNok.PowerState' -as [type])) {
            Add-Type -Namespace AksjeNok -Name PowerState -MemberDefinition '[System.Runtime.InteropServices.DllImport("kernel32.dll")] public static extern uint SetThreadExecutionState(uint flags);'
        }
        [void][AksjeNok.PowerState]::SetThreadExecutionState([uint32]2147483649)
        foreach ($Name in $Required) { Upload (Join-Path $Package $Name) $Name }
        Upload "$LocalJob\plan.json" 'plan.json'
        Upload "$KeysFolder\archive-certificate.pem" 'archive-certificate.pem'
        Upload "$KeysFolder\operator-certificate.pem" 'operator-certificate.pem'
        Write-Host '[1/5] Samlet forhandskontroll, ingen kontohandlinger'
        Remote "python $RemoteJob/render_batch.py preflight --work $RemoteJob"
        Download 'preflight-result.json'
        $Preflight = Get-Content "$LocalJob\preflight-result.json" -Raw | ConvertFrom-Json
        $Preflight | Format-List
        $Drive = Get-PSDrive -Name ([System.IO.Path]::GetPathRoot($LocalJob).Substring(0,1))
        if ($Drive.Free -lt (2 * [double]$Preflight.source_bytes)) { throw 'For lite ledig plass til uavhengig arkivkopi paa PC-en' }
        Write-Host 'Kontroller at alle relevante tjenester, gamle/eksterne lagre er inkludert, cron er suspendert og alle skrivere er stanset. Hold app-faner lukket. Ikke bekreft hvis dette er ukjent.'
        $Ack = Read-Host 'Skriv KILDENE ER KOMPLETTE OG SKRIVERNE STANSET'
        if ($Ack -ne 'KILDENE ER KOMPLETTE OG SKRIVERNE STANSET') { throw 'Ingen arkivering uten bekreftet kildedekning og skriverstans' }
        Write-NewUtf8 "$LocalJob\source-attestation.json" (@{inventory_sha256=$Preflight.inventory_sha256;all_sources_reviewed_and_writers_stopped=$true;actor=$Identity;at=[DateTime]::UtcNow.ToString('o')} | ConvertTo-Json)
        Upload "$LocalJob\source-attestation.json" 'source-attestation.json'
        Write-Host '[2/5] Installerer isolerte PG18-verktoy i arbeidsmappen, ingen ny betalt ressurs'
        Remote "NOK_WORK=$RemoteJob bash $RemoteJob/install_pg18.sh"
        Upload "$KeysFolder\archive-key.pem" 'archive-key.pem'
        Remote "chmod 600 $RemoteJob/archive-key.pem"
        Write-Host '[3/5] Komplett kryptert sluttarkiv'
        Remote "python $RemoteJob/render_batch.py archive --work $RemoteJob"
        Download 'archive-receipt.json'
        $Receipt = Get-Content "$LocalJob\archive-receipt.json" -Raw | ConvertFrom-Json
        $ArchiveId = [string]$Receipt.archive_id
        if ($ArchiveId -notmatch '^[A-Za-z0-9-]+$') { throw 'Ugyldig arkiv-ID' }
        $Archives = Join-Path $LocalJob 'independent-archives'
        New-Item -ItemType Directory $Archives | Out-Null
        Native 'scp.exe' ($ScpOptions + @('-r',"${RenderSsh}:$RemoteJob/archives/$ArchiveId",$Archives))
        $Cipher = Join-Path $Archives "$ArchiveId\archive.cms"
        $CipherHash = (Get-FileHash $Cipher -Algorithm SHA256).Hash.ToLowerInvariant()
        if ($CipherHash -ne $Receipt.cipher_sha256) { throw 'Uavhengig Windows-kopi har feil kontrollsum' }
        Native $OpenSSL @('cms','-decrypt','-binary','-inform','DER','-in',$Cipher,'-recip',"$KeysFolder\archive-certificate.pem",'-inkey',"$KeysFolder\archive-key.pem",'-out','NUL')
        $CopyProof = @{archive_id=$ArchiveId;cipher_sha256=$CipherHash;verified_windows_copy=$true;local_decryption_verified=$true;copy_kind='WINDOWS_OFFSITE_VERIFIED_SCP_COPY';verified_at=[DateTime]::UtcNow.ToString('o')}
        Write-NewUtf8 "$LocalJob\windows-copy-proof.json" ($CopyProof | ConvertTo-Json)
        Upload "$LocalJob\windows-copy-proof.json" 'windows-copy-proof.json'
        Write-Host '[4/5] Fysisk restore paa eksisterende arbeidsplass og opprettelse av INAKTIVE DRAFT-kontoer'
        Remote "python $RemoteJob/render_batch.py restore-drafts --work $RemoteJob"
        Write-Host '[5/5] Henter faktiske kvitteringer og samlet status'
        foreach ($Name in @('final-result.json','restore-result.json','prepared.json','account-settings-for-review.json','inventory.json','source-databases.json','config.json')) { Download $Name }
        $Result = Get-Content "$LocalJob\final-result.json" -Raw | ConvertFrom-Json
        $Result | Format-List
        Write-Host "Ferdig med denne avgrensede kjoringen. Privat arkiv og kvitteringer: $LocalJob"
        Write-Host 'Ingen konto er aktivert. B er ikke klar: faktiske kostnader, livebevis og signerte grants maa dokumenteres.'
        Write-Host 'Del bare final-result.json som status, ikke arkivet, private nokler eller hele produksjonskonfigurasjonen.'
    }
    catch {
        $Failure = $_
        foreach ($Stage in @('preflight','archive','restore-drafts')) {
            try { & scp.exe @ScpOptions "${RenderSsh}:$RemoteJob/STOPPED-$Stage.json" "$LocalJob\STOPPED-$Stage.json" 2>$null } catch { }
        }
        Write-Host "Jobben er stoppet uten aktivering. Eventuelle private STOPPED-kvitteringer ligger i $LocalJob"
        throw $Failure
    }
    finally {
        if ('AksjeNok.PowerState' -as [type]) { [void][AksjeNok.PowerState]::SetThreadExecutionState([uint32]2147483648) }
        if ($RemoteCreated) {
            try {
                & ssh.exe @SshOptions $RenderSsh "python $RemoteJob/render_batch.py cleanup --work $RemoteJob"
                if ($LASTEXITCODE -ne 0) { throw 'Cleanup failed' }
            } catch { Write-Warning 'Midlertidig arkivnokkel/restore-prosess maa kontrolleres i den private jobbmappen. Originalnokkelen paa PC-en bevares.' }
        }
        Write-Host "Jobbmappe: $LocalJob"
        Write-Host 'Ingen automatisk ny start, sletting av historikk eller overgang til annen periode.'
    }
}
Start-AksjeOperator
